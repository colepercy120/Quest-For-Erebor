# An artifact has changed ownership
# root = the artifact
# scope:owner = the person who now holds the artifact
# scope:old_owner = the person who used to hold the artifact
lotr_on_artifact_changed_owner = {
	on_actions = { qfe_on_artifact_changed_owner }
}

qfe_on_artifact_changed_owner = {
	effect = {
		save_scope_as = artifact

		

		if = {
			limit = {
				OR = {
					has_variable = one_ring
				}
				sauron_is_alive = yes
			}
			scope:owner = {
				trigger_event = qfe.0100 # Ring modifier switch event

				if = {
					limit = {
						cannot_be_killed = no
						NOT = { is_elf = yes }
						NOT = { is_dwarf = yes }
						any_character_artifact = {
							has_variable = one_ring # The Three aren't touched by Sauron, and will have less personality effects
						}
					}
					trigger_event = {
						on_action = nazgul_personality # Ring madness for Human and Orc
						days = { 180 225 }
					}
				}
			}
		}
	}
}
