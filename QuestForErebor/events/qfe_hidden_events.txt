qfe.0100 = { # Ring of Power modifier switch
	hidden  = yes
	
	trigger = {
		any_character_artifact = { 
			has_variable = one_ring 
		}
	}
	
	immediate = {
		every_character_artifact = {
			limit = { 
				has_variable = one_ring 
			}
			remove_one_ring_modifiers = yes
			
			
			if = {
				limit = { # Elves and Istari
					root = {
						OR = {
							is_elf = yes
							is_istari = yes
						}
					}
				}
				add_artifact_modifier = one_ring_elves_modifier
				set_artifact_rarity_illustrious = yes
			}
			
			else_if = {
				limit = { root = { is_sauron = yes } }
				add_artifact_modifier = one_ring_sauron_modifier
				if = { #A little additional layer of corruption
					limit = { NOT = { has_variable = corrupted_ring } }
					set_variable = corrupted_ring
				}
				set_artifact_rarity_illustrious = yes
				root = {
					trigger_event = qfe_situation.003
				}
			}
			
			else = { add_artifact_modifier = one_ring_men_modifier }
		}
	}
