qfe_rhovanion.003 = { # Start War
	content_source = realms_dlc
	type = character_event
	title = qfe_rhovanion.003.t
	desc = qfe_rhovanion.003.desc
	theme = war
	
	left_portrait = {
		character = root
		animation = aggressive_sword
	}
	right_portrait = {
		character = title:e_gondor_steward.holder
		animation = aggressive_sword
	}
	override_background = {
		reference = wilderness_steppe
	}
	
	
	immediate = {
	
	}

	option = {
		name = qfe_rhovanion.003.a
		current_travel_plan = {
			complete_travel_plan = yes
		}
		start_war = {
			casus_belli = rhovanion_invasion_war 
			target = title:e_golden_realm_rhun.holder 
		
			target_title = title:e_rhovanion
		}
		add_legitimacy = 400
		every_character_war = {
				
			add_attacker = title:e_gondor_steward.holder
			
		}
		every_county_in_region = {
			region = custom_plains_rhovanion
			limit = {
				holder = {
					faith_is_good = no
					is_independent_ruler = yes
					NOT = {
						title:k_wastelands.holder = this
					}
					culture = {
						has_same_culture_heritage = culture:magriag
					}
				}
			}
			holder = {
				save_scope_as = target1
				root = {
					every_character_war = {
						add_defender = scope:target1
					}
				}
				
			}
		}
		every_independent_ruler = {
			limit = {
				culture = culture:rhovanorim
				NOT = {
					this = root
				}
			}
			
			trigger_event = qfe_rhovanion.004
			
		}
		every_war_enemy = {
			trigger_event = qfe_rhovanion.005
		}
		title:e_mordor.holder = {
			trigger_event = qfe_rhovanion.006
		}
	}
	after = {
		
	}
}
