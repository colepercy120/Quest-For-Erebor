qfe_intro.012 = { # Introduction to Aragorn
	content_source = realms_dlc
	type = character_event
	title = qfe_intro.012.t
	desc = qfe_intro.012.desc
	theme = stewardship_duty_focus

	override_background = {
		reference = throne_room_rivendell
	}
	
	right_portrait = {
		character = character:lineofelrond1
		animation = happy_teacher
	}
	left_portrait = {
		character = character:lineisildur42
		animation = clutching_doll
	}
	
	immediate = {
		play_music_cue = mx_mood_call_24
		hidden_effect = {
			random_artifact = {
				limit = {
					has_variable = shards_of_narsil
				}
				set_owner = character:lineofelrond1
			}	
			random_artifact = {
				limit = {
					has_variable = ring_barahir
				}
				set_owner = character:lineofelrond1
			}	
		}
	}
	
	option = { # From the ashes a fire shall be woken
		name = qfe_intro.012.a
		#custom_tooltip = qfe_intro.012.a.tt
		title:e_arnor = { # Every landed Dunedain Ranger in Arnor gets the chance to swear fealty.
			every_de_jure_county_holder = {
				limit = {
					is_independent_ruler = yes
					culture = culture:dunedain_ranger
					location.empire = title:e_arnor
					NOT = { has_title = title:d_northern_dunedain }
				}
				trigger_event = {
					id = rangers.0101
					days = 1
				}
			}
		}
		trigger_event = {
			id = qfe_intro.013
			years = 12
		}
	}
}
